# Wonders in Civs - monument journal entries
# One journal entry per buildable wonder. The entry is shown to any country
# that owns the wonder's state region, and completes once the building exists.

je_wic_aerhanbula = {
	icon = "gfx/interface/icons/building_icons/aerhanbula.dds"
	group = je_group_wic_wonders

	is_shown_when_inactive = {
		any_scope_state = {
			state_region = s:STATE_UPPER_ANDALUSIA
		}
	}

	possible = {
		any_scope_state = {
			state_region = s:STATE_UPPER_ANDALUSIA
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_wic_aerhanbula_status_building
				trigger = {
					any_scope_state = {
						state_region = s:STATE_UPPER_ANDALUSIA
						any_scope_building = {
							is_building_type = building_aerhanbula
						}
					}
				}
			}
			triggered_desc = {
				desc = je_wic_aerhanbula_status_none
				trigger = { always = yes }
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = je_wic_aerhanbula_complete_tt
			any_scope_state = {
				state_region = s:STATE_UPPER_ANDALUSIA
				has_building = building_aerhanbula
			}
		}
	}

	on_complete = {
		trigger_event = {
			id = wic_wonders.1
			days = 0
			popup = yes
		}
	}

	invalid = {
		NOT = {
			any_scope_state = {
				state_region = s:STATE_UPPER_ANDALUSIA
			}
		}
	}

	fail = {
		custom_tooltip = {
			text = je_wic_aerhanbula_fail_tt
			NOT = {
				any_scope_state = {
					state_region = s:STATE_UPPER_ANDALUSIA
					has_building = building_aerhanbula
				}
			}
			s:STATE_UPPER_ANDALUSIA = {
				any_scope_state = {
					has_building = building_aerhanbula
				}
			}
		}
	}

	on_fail = {
		trigger_event = {
			id = wic_wonders_fail.1
			days = 0
			popup = yes
		}
	}

	event_outcome_failed_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_wic_aerhanbula_failed
			}
		}
	}

	event_outcome_failed_effect_desc = {
		header = wic_wonders_fail.shared
		effect = {
			add_modifier = {
				name = wic_wonder_lost_race
				days = moderately_long_modifier_time
			}
		}
	}


	event_outcome_completed_effect_desc = {
		header = wic_wonders.shared.a
		effect = {
			add_modifier = {
				name = wic_wonder_aerhanbula_glory
				days = normal_modifier_time
			}
		}
	}

	event_outcome_completed_effect_desc = {
		header = wic_wonders.shared.b
		effect = {
			add_treasury = 30000
			random_scope_state = {
				limit = {
					state_region = s:STATE_UPPER_ANDALUSIA
					has_building = building_aerhanbula
				}
				add_modifier = {
					name = wic_wonder_aerhanbula_profit
					days = normal_modifier_time
				}
			}
		}
	}

	weight = 900
}
