namespace = wic_wonders

# Completion events for the Wonders in Civs journal entries.
# Each event is fired by the on_complete of its journal entry.

wic_wonders.1 = {
	type = country_event
	placement = scope:wic_wonder_state

	duration = 3

	title = wic_wonders.1.t
	desc = wic_wonders.1.d
	flavor = wic_wonders.1.f

	event_image = {
		video = "unspecific_world_fair"
	}

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	immediate = {
		random_scope_state = {
			limit = {
				state_region = s:STATE_UPPER_ANDALUSIA
				has_building = building_aerhanbula
			}
			save_scope_as = wic_wonder_state
		}
	}

	option = {
		name = wic_wonders.shared.a
		default_option = yes
		add_modifier = {
			name = wic_wonder_aerhanbula_glory
			days = normal_modifier_time
		}
	}

	option = {
		name = wic_wonders.shared.b
		add_treasury = 30000
		scope:wic_wonder_state = {
			add_modifier = {
				name = wic_wonder_aerhanbula_profit
				days = normal_modifier_time
			}
		}
	}
}
